\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{positioning, arrows.meta}

\begin{document}
\begin{tikzpicture}[
    box/.style={draw, rounded corners, align=center, minimum height=0.9cm, inner sep=6pt, font=\small},
    lab/.style={font=\scriptsize, text=black!60},
    arr/.style={-{Stealth[length=6pt]}, thick, black!55}
  ]

  % original sample
  \node[box, fill=blue!8, minimum width=3.4cm] (orig) {$X_1, \dots, X_n \sim F$ \\ \scriptsize original sample};

  % resampled datasets
  \node[box, fill=orange!8, minimum width=3.4cm, right=2.6cm of orig, yshift=2.1cm] (r1) {$X_1^{*(1)}, \dots, X_n^{*(1)}$};
  \node[box, fill=orange!8, minimum width=3.4cm, below=0.35cm of r1] (r2) {$X_1^{*(2)}, \dots, X_n^{*(2)}$};
  \node[lab, below=0.15cm of r2] (dots1) {$\vdots$};
  \node[box, fill=orange!8, minimum width=3.4cm, below=0.55cm of dots1] (rB) {$X_1^{*(B)}, \dots, X_n^{*(B)}$};

  % statistics
  \node[box, fill=green!8, minimum width=1.7cm, right=2.4cm of r1] (t1) {$T_n^{*(1)}$};
  \node[box, fill=green!8, minimum width=1.7cm, right=2.4cm of r2] (t2) {$T_n^{*(2)}$};
  \node[lab, right=2.4cm of dots1] (dots2) {$\vdots$};
  \node[box, fill=green!8, minimum width=1.7cm, right=2.4cm of rB] (tB) {$T_n^{*(B)}$};

  % bootstrap distribution
  \node[box, fill=blue!8, minimum width=3.2cm, minimum height=2.6cm, right=2.6cm of t2, yshift=-0.05cm] (dist) {
    \begin{tikzpicture}[scale=0.75]
      \draw[black!50] (0,0) -- (2.6,0);
      \foreach \x/\h in {0.1/0.25, 0.4/0.55, 0.7/0.95, 1.0/1.35, 1.3/1.55, 1.6/1.3, 1.9/0.9, 2.2/0.45}
        \draw[fill=blue!45, draw=blue!70] (\x,0) rectangle ++(0.25,\h);
    \end{tikzpicture} \\[-2pt]
    \scriptsize bootstrap distr.\ of $T_n^*$
  };

  % arrows: original -> resamples
  \draw[arr] (orig.east) -- ++(0.5,0) |- (r1.west);
  \draw[arr] (orig.east) -- ++(0.5,0) |- (r2.west);
  \draw[arr] (orig.east) -- ++(0.5,0) |- (rB.west);

  % arrows: resamples -> statistics
  \draw[arr] (r1.east) -- (t1.west);
  \draw[arr] (r2.east) -- (t2.west);
  \draw[arr] (rB.east) -- (tB.west);

  % arrows: statistics -> distribution
  \draw[arr] (t1.east) -- ++(0.4,0) |- (dist.west);
  \draw[arr] (t2.east) -- (dist.west);
  \draw[arr] (tB.east) -- ++(0.4,0) |- (dist.west);

  % stage labels
  \node[lab, above=0.15cm of orig] {\scriptsize data};
  \node[lab, above=0.55cm of r1] {\scriptsize $B$ resamples (with replacement)};
  \node[lab, above=0.55cm of t1] {\scriptsize statistic per resample};
  \node[lab, above=0.15cm of dist] {\scriptsize result};

\end{tikzpicture}
\end{document}
